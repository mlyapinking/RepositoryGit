// Отчёт 3: История аренд клиента
// Имя отчёта в конфигурации: ОтчетИсторияКлиента
//
// Параметр СКД:
//   &Клиент — СправочникСсылка.Клиенты

ВЫБРАТЬ
    Договоры.Ссылка КАК НомерДоговора,
    Договоры.Дата КАК ДатаДоговора,
    Договоры.Автомобиль КАК Автомобиль,
    Договоры.Автомобиль.Наименование КАК АвтомобильНаименование,
    Договоры.ДатаНачала КАК ДатаНачала,
    Договоры.ДатаОкончания КАК ДатаОкончания,
    Договоры.ЦенаЗаСутки КАК ЦенаЗаСутки,
    Договоры.СуммаЗалога КАК СуммаЗалога,
    ЕСТЬNULL(ФактПроката.Ссылка, ЗНАЧЕНИЕ(Документ.ФактПроката.ПустаяСсылка)) КАК НомерЗакрытия,
    ЕСТЬNULL(ФактПроката.ДатаВозврата, ДАТАВРЕМЯ(1, 1, 1)) КАК ДатаВозврата,
    ЕСТЬNULL(ФактПроката.КоличествоСуток, 0) КАК КоличествоСуток,
    ЕСТЬNULL(ФактПроката.ИтоговаяСумма, 0) КАК ИтоговаяСумма,
    ВЫБОР
        КОГДА ФактПроката.Ссылка ЕСТЬ NULL
            ТОГДА "В аренде"
        ИНАЧЕ "Закрыта"
    КОНЕЦ КАК СтатусАренды
ИЗ
    Документ.ДоговАренды КАК Договоры
        ЛЕВОЕ СОЕДИНЕНИЕ Документ.ФактПроката КАК ФактПроката
        ПО ФактПроката.Договор = Договоры.Ссылка
            И ФактПроката.Проведен
ГДЕ
    Договоры.Проведен
    И Договоры.Клиент = &Клиент

УПОРЯДОЧИТЬ ПО
    ДатаДоговора
